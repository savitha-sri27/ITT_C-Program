int maxpile(int piles[],int n)
{
    int max=piles[0];
    for(int i=0;i<n;i++)
    {
        if(piles[i]>max)
        {
            max=piles[i];
        }
    }
    return max;
}
int minEatingSpeed(int* piles, int pilesSize, int h) {
   int l=1;
   int high=maxpile(piles,pilesSize);
   long long hrs=0;
   while(l<high)
   {
    int m=l+ (high - l) / 2;
    long long hrs=0;
    for(int i=0;i<pilesSize;i++)
    {hrs+=(piles[i] + m - 1) / m;
    if(hrs > h)
                break;
    }
   
   if(hrs <= h)
            high = m;
        else
            l = m + 1;
    
   }
   return l;
}
