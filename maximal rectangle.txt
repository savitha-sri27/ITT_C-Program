#include <stdio.h>
#include <stdlib.h>
#define MAX(a, b) ((a) > (b) ? (a) : (b))
int largestRectangleArea(int* heights, int heightsSize) {
    int* stack = (int*)malloc((heightsSize + 1) * sizeof(int));
    int top = -1;
    int maxArea = 0;   
    int i = 0;
    while (i <= heightsSize) {
        int currentHeight = (i == heightsSize) ? 0 : heights[i];
        
        if (top == -1 || currentHeight >= heights[stack[top]]) {
            stack[++top] = i;
            i++;
        } else {
            int heightIdx = stack[top--];
            int h = heights[heightIdx];
            int w = (top == -1) ? i : (i - stack[top] - 1);
            maxArea = MAX(maxArea, h * w);
        }
    }   
    free(stack);
    return maxArea;
}
int maximalRectangle(char** matrix, int matrixSize, int* matrixColSize) {
    if (matrixSize == 0 || matrixColSize[0] == 0) return 0;
    
    int rows = matrixSize;
    int cols = matrixColSize[0];
    int* heights = (int*)calloc(cols, sizeof(int));
    int maxRectangle = 0;
    for (int r = 0; r < rows; r++) {
        for (int c = 0; c < cols; c++) {
            if (matrix[r][c] == '1') {
                heights[c]++;
            } else {
                heights[c] = 0;
            }
        }
        int area = largestRectangleArea(heights, cols);
        maxRectangle = MAX(maxRectangle, area);
    }   
    free(heights);
    return maxRectangle;
}
