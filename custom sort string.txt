#include <stdio.h>
#include <stdlib.h>
#include <string.h>
char* customSortString(char* order, char* s) {
    int freq[26] = {0};
    int sLen = strlen(s);
    for (int i = 0; i < sLen; i++) {
        freq[s[i] - 'a']++;
    }
    char* result = (char*)malloc((sLen + 1) * sizeof(char));
    int index = 0;
    int orderLen = strlen(order);
    for (int i = 0; i < orderLen; i++) {
        char ch = order[i];
        while (freq[ch - 'a'] > 0) {
            result[index++] = ch;
            freq[ch - 'a']--;
        }
    }
    for (int i = 0; i < 26; i++) {
        while (freq[i] > 0) {
            result[index++] = i + 'a';
            freq[i]--;
        }
    }
    result[index] = '\0';  
    return result;
}
